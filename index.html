from wkafka import WKafka

kafka = WKafka(bootstrap_servers="localhost:29092", client_id="json_app")

@kafka.consumer(topic="orders", format="json")
def on_order(msg):
    print(f"📦 Received order #{msg.value['id']}: {msg.value['item']}")

if __name__ == "__main__":
    kafka.send("orders", value={"id": 101, "item": "Laptop GPU"})
    kafka.run_consumers(block=True)

from wpipe import Pipeline, step, to_obj
from wkafka import WKafka
from pydantic import BaseModel

class StreamContext(BaseModel):
    token: str
    status: str = "pending"

@step(name="KafkaStep", version="v1.0")
class KafkaConsumerStep:
    def __init__(self):
        self.kafka = WKafka(bootstrap_servers="localhost:29092")

    @to_obj(StreamContext)
    def __call__(self, ctx: StreamContext) -> StreamContext:
        ctx.status = "processed_via_kafka"
        return ctx

if __name__ == "__main__":
    pipeline = Pipeline(pipeline_name="wkafka_pipeline", tracking_db="tracking.db")
    pipeline.set_steps([KafkaConsumerStep()])
    pipeline.run({"token": "tx_99"})
